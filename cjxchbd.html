二、STM32 蜂鸣器控制

基于 STM32F103 + ESP8266 WiFi 模块，通过 MQTT 协议接入 OneNET 平台。接收小程序下发的 buzzer_switch 指令，实现蜂鸣器间歇鸣响（响1秒/停2秒循环）。

2.1 蜂鸣器驱动 — buzzer.c / buzzer.h

/* buzzer.h — 蜂鸣器引脚宏定义 */
#define S12BUZZER_RCC   RCC_APB2Periph_GPIOA
#define S12BUZZER_PORT  GPIOA
#define S12BUZZER_PIN   GPIO_Pin_6

void Buzzer_Init(void);
void Control_Buzzer(u8 cmd);    // cmd=1 响, cmd=0 停

/* buzzer.c — 蜂鸣器 GPIO 推挽输出驱动 */
void Buzzer_Init(void)
{
    GPIO_InitTypeDef  GPIO_InitStructure;
    RCC_APB2PeriphClockCmd(S12BUZZER_RCC, ENABLE);       // 使能 GPIOA 时钟
    GPIO_InitStructure.GPIO_Pin  = S12BUZZER_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;      // 推挽输出
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(S12BUZZER_PORT, &GPIO_InitStructure);
    GPIO_ResetBits(S12BUZZER_PORT, S12BUZZER_PIN);        // 默认关闭
}

void Control_Buzzer(u8 cmd)
{
    if (cmd == 1)  GPIO_SetBits(S12BUZZER_PORT, S12BUZZER_PIN);    // 高电平 → 响
    else        GPIO_ResetBits(S12BUZZER_PORT, S12BUZZER_PIN);  // 低电平 → 停
}

2.2 主控逻辑 — main.c

/* main.c — STM32 蜂鸣器设备主程序 */
uint8_t g_buzzer_enabled = 0;        // 全局蜂鸣器使能标志（由云平台下行设置）

int main(void)
{
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    Delay_Init();                              // SysTick 初始化
    Usart3_Init(115200);                       // 串口3 — 调试打印
    Usart2_Init(115200);                       // 串口2 — 驱动 ESP8266
    ESP8266_Init();                            // WiFi 模块初始化
    OLED_Init();  LED_Init();
    Sensor_Init();  Buzzer_Init();

    /* 连接 OneNET MQTT 服务器 */
    while (ESP8266_SendCmd(ESP8266_ONENET_INFO, "CONNECT"))  delay_ms(500);
    while (OneNet_DevLink())  delay_ms(500);   // 鉴权 + MQTT CONNECT
    OneNET_Subscribe();                        // 订阅属性设置主题

    while (1)
    {
        /* 蜂鸣器间歇控制：响1秒(10 ticks) / 停2秒(20 ticks) */
        if (g_buzzer_enabled) {
            buzzer_tick++;
            if      (buzzer_tick <= 10)  Control_Buzzer(1);   // 0~1s: 响
            else if (buzzer_tick <= 30)  Control_Buzzer(0);   // 1~3s: 停
            else  buzzer_tick = 0;
        } else {
            Control_Buzzer(0);  buzzer_tick = 0;
        }

        /* 处理云平台下行数据 */
        dataPtr = ESP8266_GetIPD(0);
        if (dataPtr != NULL)  OneNet_RevPro(dataPtr);   // 解析 MQTT 消息

        /* 每 3 秒上报一次设备状态 */
        if (++send_count >= 30) { send_count = 0;  OneNet_SendData(); }

        delay_ms(100);
    }
}

2.3 OneNET MQTT 协议栈 — onenet.c（核心片段）

/* onenet.c — OneNET MQTT 连接、上报、订阅、下行解析 */
#define OneNet_Product_ID   "182uSs4N9a"
#define OneNet_Access_KEY   "8V9zxR1hO/WVs8kXUx9dzz9D/XG3d0mhUdVnezFvI8E="
#define OneNet_Device_NAME  "addp_03"

/* ── SHA1-HMAC 签名生成 Token ── */
static unsigned char OneNET_Authorization(...) {
    // 1. Base64 解码 AccessKey → 原始密钥
    BASE64_Decode(access_key_base64, ..., access_key, ...);
    // 2. 拼接签名字符串: et\nmethod\nres\ndev_name\nversion
    snprintf(string_for_signature, ..., "%d\n%s\nproducts/%s/devices/%s\n%s", ...);
    // 3. HMAC-SHA1 计算签名 → Base64 编码 → URL 转义
    hmac_sha1(key, ..., string_for_signature, ..., hmac_buf);
    BASE64_Encode(sign_buf, ..., hmac_buf, ...);
    OTA_UrlEncode(sign_buf);
    // 4. 组装 authorization 参数字符串
}

/* ── MQTT CONNECT 接入 OneNET ── */
_Bool OneNet_DevLink(void) {
    MQTT_PacketConnect(Product_ID, authorization_buf, Device_NAME, 256, 1, MQTT_QOS_LEVEL0, ...);
    ESP8266_SendData(mqttPacket._data, mqttPacket._len);   // 通过 ESP8266 发送
    // 等待 CONNACK 确认接入成功
}

/* ── 数据上报：JSON 格式 → $sys/{pid}/{dev}/thing/property/post ── */
void OneNet_SendData(void) {
    snprintf(buf, 256,
        "{\"method\":\"thing.property.post\",\"id\":\"1\",\"params\":{"
        "\"buzzer_switch\":{\"value\":%s}}}",
        g_buzzer_enabled ? "true" : "false");
    OneNET_Publish(topic, buf);
}

/* ── 订阅属性设置 & 属性查询主题 ── */
void OneNET_Subscribe(void) {
    // topic_set: $sys/{pid}/addp_03/thing/property/set
    // topic_get: $sys/{pid}/addp_03/thing/property/get
    MQTT_PacketSubscribe(MQTT_SUBSCRIBE_ID, MQTT_QOS_LEVEL0, topics, 2, &mqtt_packet);
}

/* ── 下行解析：提取 buzzer_switch 并控制蜂鸣器 ── */
void OneNet_RevPro(unsigned char *cmd) {
    // MQTT_UnPacketRecv → 判断包类型 (PUBLISH / PUBACK / SUBACK)
    // PUBLISH:  解析 topic + payload → JSON 中提取 "buzzer_switch"
    //           支持嵌套格式 {"buzzer_switch":{"value":true}} 和直接格式 {"buzzer_switch":true}
    //           设置 g_buzzer_enabled 标志位
    //           → 发布 set_reply / get_reply 回执
}
场景联动说明：当光照设备 (addp_01) 检测到光照 ≥ 3000 lux 时，通过 OneNET 服务调用向 addp_03 的 thing/property/set 主题下发 {"buzzer_switch":{"value":true}}，触发蜂鸣器间歇鸣响。

三、STM32 光照传感器 + LED 控制

STM32F103 通过 ADC 采集光照强度，支持手动/自动双模式控制 PB1 绿色 LED。自动模式下 TIM2 定时器每 100ms 根据阈值自动开关灯。

3.1 光照传感器驱动 — light.c

/* light.c — 光照传感器 ADC 采集 + Lux 换算 */
#define LIGHT_ADC_CHANNEL   ADC_Channel_7     // PA7 — ADC1 通道7
#define LIGHT_CAL_ADC1      1200              // 300 Lux 对应 ADC 值
#define LIGHT_CAL_LUX1      300
#define LIGHT_CAL_ADC2      3000              // 8000 Lux 对应 ADC 值
#define LIGHT_CAL_LUX2      8000

/* ADC 初始化：独立模式、软件触发、右对齐 */
void Light_Init(void) {
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA | RCC_APB2Periph_ADC1, ENABLE);
    // PA7 配置为模拟输入
    // ADC1: 独立模式、非扫描、非连续、软件触发、右对齐
    ADC_Cmd(ADC1, ENABLE);
    ADC_ResetCalibration(ADC1);                 // 复位校准
    while(ADC_GetResetCalibrationStatus(ADC1));
    ADC_StartCalibration(ADC1);                 // 开始校准
    while(ADC_GetCalibrationStatus(ADC1));
}

/* 10 次采样取平均值，消除噪声 */
uint16_t Light_GetAdcAverage(void) {
    uint32_t temp = 0;
    for(uint8_t t = 0; t < 10; t++) {
        ADC_SoftwareStartConvCmd(ADC1, ENABLE);
        while(!ADC_GetFlagStatus(ADC1, ADC_FLAG_EOC));  // 等待转换完成
        temp += ADC_GetConversionValue(ADC1);
    }
    return (uint16_t)(temp / 10);
}

/* ADC 值 → Lux 线性插值换算，钳位到 [0, 50000] */
uint16_t Light_GetLux(void) {
    uint16_t adc = Light_GetAdcAverage();
    float lux = (float)(adc - LIGHT_CAL_ADC1) * (LIGHT_CAL_LUX2 - LIGHT_CAL_LUX1)
              / (LIGHT_CAL_ADC2 - LIGHT_CAL_ADC1) + LIGHT_CAL_LUX1;
    if(lux < 0) lux = 0;
    if(lux > 50000) lux = 50000;
    return (uint16_t)lux;
}

3.2 主控逻辑 + TIM2 自动控制 — main.c

/* main.c — 光照设备(PB1 LED) — 手动/自动双模式 */
#define LIGHT_THRESHOLD   1000                  // 自动开灯阈值 (Lux)

u16 light = 0;
volatile u8 led_mode = 0;                 // 0=自动模式, 1=手动模式
volatile u8 manual_led_state = 0;         // 手动模式下 LED 状态

/* TIM2 初始化：72MHz / (7199+1) / (999+1) = 10Hz (100ms 中断) */
void TIM2_Light_Init(void) {
    TIM_TimeBaseStructure.TIM_Period = 999;
    TIM_TimeBaseStructure.TIM_Prescaler = 7199;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseStructure);
    TIM_ITConfig(TIM2, TIM_IT_Update, ENABLE);
    TIM_Cmd(TIM2, ENABLE);
}

/* TIM2 中断服务函数：自动模式下根据光照阈值控制 LED */
void TIM2_IRQHandler(void) {
    if(TIM_GetITStatus(TIM2, TIM_IT_Update) != RESET) {
        TIM_ClearITPendingBit(TIM2, TIM_IT_Update);
        if(led_mode == 0) {                        // 仅自动模式下生效
            if(light < LIGHT_THRESHOLD)  LED_Green_On();   // 光线暗 → 开灯
            else                       LED_Green_Off();  // 光线足 → 关灯
        }
    }
}

int main(void) {
    /* 初始化：系统时钟、串口、ESP8266、LED、OLED、光照、TIM2 */
    LED_Green_Off();

    /* 连接 OneNET + 订阅 */
    while(OneNet_DevLink())  delay_ms(500);
    OneNET_Subscribe();

    while(1) {
        Read_Sensors();                          // light = Light_GetLux()
        /* OLED 显示：光照值 + 当前模式 */
        OLED_ShowString(0, 0, "Light: ", 12);  OLED_ShowNum(48, 0, light, 5, 12);
        // Mode: AUTO / MANUAL

        /* 处理云平台下行：解析 led 字段切换模式 */
        dataPtr = ESP8266_GetIPD(0);
        if(dataPtr != NULL)  OneNet_RevPro(dataPtr);

        /* 每 1 秒上报光照数据 */
        if(++send_count >= 10) { send_count = 0;  OneNet_SendData(); }

        delay_ms(100);
    }
}
模式切换：小程序发送 led=true → STM32 进入手动模式（灯亮）；发送 led=false → STM32 切回自动模式（TIM2 根据阈值自动控灯）。

四、STM32 人体红外传感器

HC-SR501 人体红外传感器通过 GPIO 输入检测人体活动。检测到人体时绿色 LED 亮起，OLED 实时显示状态，数据每 1 秒上报 OneNET。

4.1 人体传感器驱动 — hum.c

/* hum.c — HC-SR501 人体红外传感器驱动（GPIO 浮空输入） */
#define HUMAN_GPIO_PORT    GPIOA
#define HUMAN_GPIO_PIN     GPIO_Pin_4
#define HUMAN_GPIO_RCC     RCC_APB2Periph_GPIOA

/* 初始化 PA4 为浮空输入模式 */
void Human_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    RCC_APB2PeriphClockCmd(HUMAN_GPIO_RCC, ENABLE);
    GPIO_InitStructure.GPIO_Pin  = HUMAN_GPIO_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;   // 浮空输入
    GPIO_Init(HUMAN_GPIO_PORT, &GPIO_InitStructure);
}

/* 读取 PA4 电平：高电平=有人，低电平=无人 */
uint8_t Human_Detect(void)
{
    return GPIO_ReadInputDataBit(HUMAN_GPIO_PORT, HUMAN_GPIO_PIN);
}
